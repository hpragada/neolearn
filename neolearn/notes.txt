<script>
let username = localStorage.getItem("username") || "Guest";
document.getElementById("welcome").innerText="Welcome "+username;

// ================= NOTES =================
const noteInput = document.getElementById("noteInput");
const summaryOutput = document.getElementById("summaryOutput");
const notesList = document.getElementById("notesList");
const pdfFileInput = document.getElementById("pdfFile");

// PDF text extraction
async function extractPDFText(file){
    const arrayBuffer = await file.arrayBuffer();
    const pdf = await pdfjsLib.getDocument({data: arrayBuffer}).promise;
    let fullText = '';
    for(let i=1;i<=pdf.numPages;i++){
        const page = await pdf.getPage(i);
        const content = await page.getTextContent();
        const pageText = content.items.map(item=>item.str).join(' ');
        fullText += pageText+'\n\n';
    }
    return fullText;
}

// AI Summarization (simulate or call backend)
async function summarizeText(text){
    try{
        let res = await fetch("http://localhost:5000/summarize",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({text})});
        let data = await res.json();
        return data.summary;
    }catch(e){return "❌ Error summarizing note!";}
}

// AI Subject Detection (simulate)
function detectSubject(text){
    const t=text.toLowerCase();
    if(t.includes("dfa")||t.includes("nfa")||t.includes("regex")) return "FLAT";
    if(t.includes("sort")||t.includes("tree")||t.includes("graph")) return "DAA";
    return "Aptitude";
}

// Save note as PDF
document.getElementById("saveNoteBtn").addEventListener("click", async ()=>{
    const text = noteInput.value.trim();
    if(!text) return alert("No text to save!");

    // Auto detect subject
    const subject = detectSubject(text);
    document.getElementById("subject").value = subject;

    // Create PDF
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF();
    const lines = doc.splitTextToSize(text, 180);
    doc.text(lines, 15, 20);
    doc.save(`${subject}_Note.pdf`);

    // Add to UI
    const noteCard = document.createElement("div");
    noteCard.className="note-card";
    noteCard.innerHTML=`<strong>${subject}:</strong> <p>${text}</p>`;
    notesList.prepend(noteCard);
    noteInput.value="";
    summaryOutput.style.display="none";
});

// Summarize Note
document.getElementById("summarizeNoteBtn").addEventListener("click", async ()=>{
    const text = noteInput.value.trim();
    if(!text) return alert("No text to summarize!");
    summaryOutput.style.display="block";
    summaryOutput.innerHTML="🌀 Summarizing...";
    const summary = await summarizeText(text);
    summaryOutput.innerHTML = `<p>${summary}</p><button onclick="copySummary()" style="margin-top:10px;padding:8px 12px;border:none;border-radius:8px;background:#00f5c4;color:black;cursor:pointer;">📋 Copy Summary</button>`;
});

function copySummary(){
    navigator.clipboard.writeText(summaryOutput.querySelector("p").innerText);
    alert("Summary copied ✅");
}

// Upload PDF and auto summarize + detect subject
pdfFileInput.addEventListener("change", async ()=>{
    const file = pdfFileInput.files[0];
    if(!file) return alert("Select a PDF first!");
    const text = await extractPDFText(file);
    if(!text.trim()) return alert("PDF has no readable text!");
    noteInput.value = text;

    // Detect subject automatically
    const subject = detectSubject(text);
    document.getElementById("subject").value = subject;

    // Auto summarize
    await document.getElementById("summarizeNoteBtn").click();
});
</script>
